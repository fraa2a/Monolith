cmake_minimum_required(VERSION 3.20)
project(monolith_phase1_tests LANGUAGES CXX)
# Standalone: root configuration requires Windows/FFmpeg/vcpkg. No fetched deps.
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_executable(replay_packet_ring_test replay_packet_ring_test.cpp)
target_include_directories(replay_packet_ring_test PRIVATE ../../libs)
add_test(NAME replay_packet_ring COMMAND replay_packet_ring_test)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_package(Threads REQUIRED)
    find_package(nlohmann_json CONFIG REQUIRED)
    add_executable(ipc_server_test ipc_server_test.cpp)
    target_include_directories(ipc_server_test PRIVATE win32-shim)
    target_link_libraries(ipc_server_test PRIVATE Threads::Threads nlohmann_json::nlohmann_json)
    add_test(NAME ipc_server COMMAND ipc_server_test)
    set_tests_properties(ipc_server PROPERTIES TIMEOUT 120 RUN_SERIAL TRUE)
endif()

add_executable(pcm_input_test pcm_input_test.cpp)
target_include_directories(pcm_input_test PRIVATE ../../libs)
add_test(NAME pcm_input COMMAND pcm_input_test)
