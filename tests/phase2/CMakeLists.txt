cmake_minimum_required(VERSION 3.20)
project(monolith_phase2_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
find_package(PkgConfig REQUIRED)
pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil)
find_package(Threads REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
add_library(media_under_test STATIC ../../libs/encoding/trim.cpp ../../libs/encoding/mux_common.cpp
    ../../libs/disk-segments/disk_segments.cpp ../../libs/replay-buffer/replay_buffer.cpp)
target_include_directories(media_under_test PUBLIC ../../libs PRIVATE win32-shim)
target_compile_definitions(media_under_test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(media_under_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(media_under_test PUBLIC PkgConfig::AV Threads::Threads)
add_executable(media_test media_test.cpp)
target_link_libraries(media_test PRIVATE media_under_test nlohmann_json::nlohmann_json)
target_compile_options(media_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_options(media_test PRIVATE -Wl,--wrap=av_write_frame)
add_test(NAME media COMMAND media_test ${CMAKE_CURRENT_BINARY_DIR}/fixtures)
set_tests_properties(media PROPERTIES TIMEOUT 120 FIXTURES_SETUP media_fixtures)

find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME media_timeline COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/verify_timeline.py ${CMAKE_CURRENT_BINARY_DIR}/fixtures)
set_tests_properties(media_timeline PROPERTIES DEPENDS media FIXTURES_REQUIRED media_fixtures TIMEOUT 120)
