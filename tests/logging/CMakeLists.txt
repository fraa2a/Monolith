cmake_minimum_required(VERSION 3.20)
project(monolith_logging_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
enable_testing()
find_package(Threads REQUIRED)
add_executable(log_writer_test log_writer_test.cpp ../../libs/logging/logging.cpp)
target_include_directories(log_writer_test PRIVATE ../../libs win32-shim)
target_compile_options(log_writer_test PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(log_writer_test PRIVATE Threads::Threads)
add_test(NAME log_writer COMMAND log_writer_test)
