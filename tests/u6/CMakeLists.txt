cmake_minimum_required(VERSION 3.20)
project(monolith_u6_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
enable_testing()
find_package(PkgConfig REQUIRED)
pkg_check_modules(SQLITE REQUIRED IMPORTED_TARGET sqlite3)
add_executable(storage_mutation_test storage_mutation_test.cpp ../../libs/storage/storage.cpp)
target_include_directories(storage_mutation_test PRIVATE ../../libs win32-shim)
target_compile_options(storage_mutation_test PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(storage_mutation_test PRIVATE PkgConfig::SQLITE)
add_test(NAME storage_mutation COMMAND storage_mutation_test)
